/*RENDER_START*/
  function renderMaterials(){ const list=$('[data-material-list]'); if(!list) return; list.innerHTML=state.materials.length?state.materials.map(row=>`<tr><td>${row.name}</td><td>${row.quantity}</td><td>${row.dueDate||'-'}</td><td><span class="status status-material-${row.status}">${row.status==='received'?'입고완료':'구매중'}</span></td><td><div class="row-actions"><button class="small-button" data-material-edit="${row.id}">수정</button><button class="small-button" data-material-delete="${row.id}">삭제</button></div></td></tr>`).join(''):'<tr><td colspan="5" class="empty-state">등록된 자재가 없습니다.</td></tr>'; }
  function renderSpares(){ const list=$('[data-spare-list]'); if(!list) return; list.innerHTML=state.spares.length?state.spares.map(row=>`<tr><td>${row.name}</td><td>${row.quantity}</td><td><span class="status status-spare-${row.repairStatus==='수리 완료'?'complete':row.repairStatus==='수리 진행중'?'progress':'needed'}">${row.repairStatus}</span></td><td>${row.memo||'-'}</td><td><div class="row-actions"><button class="small-button" data-spare-edit="${row.id}">수정</button><button class="small-button" data-spare-delete="${row.id}">삭제</button></div></td></tr>`).join(''):'<tr><td colspan="5" class="empty-state">등록된 예비품이 없습니다.</td></tr>'; }
  function mergeByName(target, rows, kind){ let count=0; rows.forEach(row=>{ const existing=target.find(item=>item.name===row.name); if(existing) Object.assign(existing,row); else target.push({id:`${kind}-${Date.now()}-${Math.random().toString(16).slice(2)}`,...row}); count++; }); return count; }
/*RENDER_END*/
/*FORMS_START*/
  function openMaterialForm(id){ const row=state.materials.find(item=>item.id===id); openDialog(row?'자재 수정':'자재 등록',`<form data-material-form data-id="${row?.id||''}"><label>자재명<input name="name" required value="${row?.name||''}" /></label><label>수량<input name="quantity" type="number" min="0" required value="${row?.quantity??''}" /></label><label>납기일<input name="dueDate" type="date" value="${row?.dueDate||''}" /></label><label>상태<select name="status"><option value="purchasing" ${row?.status!=='received'?'selected':''}>구매중</option><option value="received" ${row?.status==='received'?'selected':''}>입고완료</option></select></label><div class="dialog-actions"><button type="button" class="button secondary" data-action="close-dialog">취소</button><button class="button primary">저장</button></div></form>`); }
  function openSpareForm(id){ const row=state.spares.find(item=>item.id===id); openDialog(row?'예비품 수정':'예비품 등록',`<form data-spare-form data-id="${row?.id||''}"><label>예비품명<input name="name" required value="${row?.name||''}" /></label><label>수량<input name="quantity" type="number" min="0" required value="${row?.quantity??''}" /></label><label>수리상태<select name="repairStatus"><option ${row?.repairStatus==='수리 필요'||!row?'selected':''}>수리 필요</option><option ${row?.repairStatus==='수리 진행중'?'selected':''}>수리 진행중</option><option ${row?.repairStatus==='수리 완료'?'selected':''}>수리 완료</option></select></label><label>메모<textarea name="memo">${row?.memo||''}</textarea></label><div class="dialog-actions"><button type="button" class="button secondary" data-action="close-dialog">취소</button><button class="button primary">저장</button></div></form>`); }
/*FORMS_END*/
/*UPLOAD_START*/
  document.addEventListener('change',(event)=>{ if(event.target.matches('[data-material-upload]')||event.target.matches('[data-spare-upload]')){ const file=event.target.files?.[0]; if(!file) return; const reader=new FileReader(); reader.onload=()=>{ const isMaterial=event.target.matches('[data-material-upload]'); const rows=isMaterial?parseMaterialsCsv(String(reader.result||'')):parseSparesCsv(String(reader.result||'')); const count=mergeByName(isMaterial?state.materials:state.spares,rows,isMaterial?'material':'spare'); saveState(); render(); toast(`${count}개 항목을 불러왔습니다.`); event.target.value=''; }; reader.readAsText(file,'UTF-8'); return; } });
/*UPLOAD_END*/
/*SUBMIT_START*/
  document.addEventListener('submit',(event)=>{ if(event.target.matches('[data-material-form]')){ event.preventDefault(); const data=new FormData(event.target); const name=String(data.get('name')||'').trim(); const quantity=Number(data.get('quantity')); if(!name||!Number.isFinite(quantity)){toast('자재명과 수량을 입력하세요.',true);return;} const id=event.target.dataset.id; const row={name,quantity,dueDate:String(data.get('dueDate')||''),status:String(data.get('status')||'purchasing')}; const existing=state.materials.find(item=>item.id===id); if(existing) Object.assign(existing,row); else state.materials.push({id:`material-${Date.now()}`,...row}); saveState(); $('[data-dialog]').close(); render(); toast('자재를 저장했습니다.'); return; } if(event.target.matches('[data-spare-form]')){ event.preventDefault(); const data=new FormData(event.target); const name=String(data.get('name')||'').trim(); const quantity=Number(data.get('quantity')); if(!name||!Number.isFinite(quantity)){toast('예비품명과 수량을 입력하세요.',true);return;} const id=event.target.dataset.id; const row={name,quantity,repairStatus:String(data.get('repairStatus')||'수리 필요'),memo:String(data.get('memo')||'').trim()}; const existing=state.spares.find(item=>item.id===id); if(existing) Object.assign(existing,row); else state.spares.push({id:`spare-${Date.now()}`,...row}); saveState(); $('[data-dialog]').close(); render(); toast('예비품을 저장했습니다.'); return; } });
/*SUBMIT_END*/
