/*MATERIALS_START*/
  function renderMaterials(){ const list=$('[data-material-list]'); if(!list) return; const rows=sortByLatestDate(state.materials,row=>row.dueDate); list.innerHTML=rows.length?rows.map(row=>`<tr><td>${row.name}</td><td>${row.quantity}</td><td>${row.dueDate||'-'}</td><td><span class="status status-material-${row.status}">${row.status==='received'?'입고완료':'구매중'}</span></td><td><div class="row-actions"><button class="small-button" data-material-edit="${row.id}">수정</button><button class="small-button" data-material-delete="${row.id}">삭제</button></div></td></tr>`).join(''):'<tr><td colspan="5" class="empty-state">등록된 자재가 없습니다.</td></tr>'; }
/*MATERIALS_END*/
/*SPARES_START*/
  function renderSpares(){ const list=$('[data-spare-list]'); if(!list) return; const rows=sortByLatestDate(state.spares,row=>row.repairStatus==='수리 완료'?row.completedDate:row.repairStatus==='수리 진행중'?row.startDate:''); list.innerHTML=rows.length?rows.map(row=>{ const date=row.repairStatus==='수리 완료'?row.completedDate:row.repairStatus==='수리 진행중'?row.startDate:''; return `<tr><td>${row.name}</td><td>${row.quantity}</td><td><span class="status status-spare-${row.repairStatus==='수리 완료'?'complete':row.repairStatus==='수리 진행중'?'progress':'needed'}">${row.repairStatus}</span></td><td>${date?`<span class="date-label">${row.repairStatus==='수리 완료'?'완료일':'시작일'} ${date}</span><br>`:''}${row.memo||'-'}</td><td><div class="row-actions"><button class="small-button" data-spare-edit="${row.id}">수정</button><button class="small-button" data-spare-delete="${row.id}">삭제</button></div></td></tr>`; }).join(''):'<tr><td colspan="5" class="empty-state">등록된 예비품이 없습니다.</td></tr>'; }
  function renderSpareMonitor(){ const monitor=state.spareMonitor||{completed:0,progress:0,needed:0}; ['completed','progress','needed'].forEach(key=>{ const node=$(`[data-spare-monitor-value="${key}"]`); if(node) node.textContent=monitor[key]??0; }); }
/*SPARES_END*/
/*FORM_START*/
  function openSpareForm(id){ const row=state.spares.find(item=>item.id===id); openDialog(row?'예비품 수정':'예비품 등록',`<form data-spare-form data-id="${row?.id||''}"><label>예비품명<input name="name" required value="${row?.name||''}" /></label><label>수량<input name="quantity" type="number" min="0" required value="${row?.quantity??''}" /></label><label>수리상태<select name="repairStatus"><option ${row?.repairStatus==='수리 필요'||!row?'selected':''}>수리 필요</option><option ${row?.repairStatus==='수리 진행중'?'selected':''}>수리 진행중</option><option ${row?.repairStatus==='수리 완료'?'selected':''}>수리 완료</option></select></label><label>수리 시작일<input name="startDate" type="date" value="${row?.startDate||''}" /></label><label>수리 완료일<input name="completedDate" type="date" value="${row?.completedDate||''}" /></label><label>메모<textarea name="memo">${row?.memo||''}</textarea></label><div class="dialog-actions"><button type="button" class="button secondary" data-action="close-dialog">취소</button><button class="button primary">저장</button></div></form>`); }
/*FORM_END*/
